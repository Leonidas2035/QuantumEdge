### [А] Контекст проблеми
Офіційна специфікація Google Gemini 4 API (листопад 2026 р.) фіксує три жорсткі обмеження кешування, що відкидають застарілі патерни Gemini 2.51:
- Explicit Caching (cachedContents) не підтримується на Gemini 4 (повертає HTTP 400); кешування є виключно автоматичним Implicit Caching ($0 за запис і зберігання, знижка 95% на хіт), а cache-control із TTL анонсовано лише з 12 січня 2027 року (із тарифікацією запису 1.25×)6.

moving_block_bootstrap_noninferiority(    paired_diff_bps: np.ndarray,    *,    block_len: int = 5,    n_boot: int = 5_000,    alpha: float = 0.05,    margin_bps: float = -2.0,    seed: int = 42,) -> dict[str, Any]:    rng = np.random.default_rng(seed)    n = len(paired_diff_bps)    if n < block_len:        raise ValueError(f"Потрібно >= {block_len} парних рішень (отримано {n}).")    n_blocks = int(np.ceil(n / block_len))    max_start = n - block_len + 1    boot_means = np.empty(n_boot, dtype=np.float64)    for b in range(n_boot):        starts = rng.integers(0, max_start, size=n_blocks)        sample = np.concatenate([paired_diff_bps[s : s + block_len] for s in starts])[:n]        boot_means[b] = float(np.mean(sample))    ci_low = float(np.percentile(boot_means, 100.0 * alpha))    ci_high = float(np.percentile(boot_means, 100.0 * (1.0 - alpha)))    return {        "n_paired": n,        "mean_diff_bps": round(float(np.mean(paired_diff_bps)), 3),        "ci_low_95_one_sided_bps": round(ci_low, 3),        "ci_high_bps": round(ci_high, 3),        "margin_bps": margin_bps,        "passed": ci_low > margin_bps,    }def compute_hold_quality(hold_records: list[dict[str, Any]], opp_threshold_bps: float = 30.0) -> float:    if not hold_records:        return 1.0    justified = sum(        1        for r in hold_records        if bool(r.get("has_open_on_track"))        or float(r.get("realized_best_net_opp_bps", 0.0)) < opp_threshold_bps    )    return round(justified / len(hold_records), 4)def evaluate_matrix(ledger_db: Path, replay_json: Path, baseline_iecu_per_dec: float = 190_000.0) -> dict[str, Any]:    ledger = UsageLedger(ledger_db)    stats = ledger.summarize()    ledger.close()    replay = json.loads(replay_json.read_text(encoding="utf-8"))    paired_diffs = np.asarray(replay["paired_pnl_diff_bps"], dtype=np.float64)    ni_res = moving_block_bootstrap_noninferiority(paired_diffs)    hold_q = compute_hold_quality(replay.get("hold_records") or [])    active_trade_share = float(replay.get("active_trade_share", 0.0))    n_dec = max(int(replay.get("total_decisions", 1)), 1)    iecu_per_dec = float(stats.get("sum_iecu", 0.0)) / n_dec    token_speedup = baseline_iecu_per_dec / max(iecu_per_dec, 1.0)    gates = {        "token_efficiency_ge_10x": token_speedup >= 10.0,        "reliability_failure_lt_1pct": float(stats.get("failure_rate", 1.0)) < 0.01,        "native_usage_coverage_ge_99pct": float(stats.get("native_usage_coverage", 0.0)) >= 0.99,        "cache_token_fraction_ge_50pct": float(stats.get("cache_token_fraction", 0.0)) >= 0.50,        "latency_p95_le_18s": (stats.get("latency_imputed_p95_ms") or 999_999.0) <= 18_000.0,        "noninferiority_passed": ni_res["passed"],        "hold_quality_ge_75pct": hold_q >= 0.75,        "anti_hold_active_share_15_to_45pct": 0.15 <= active_trade_share <= 0.45,        "zero_risk_violations": int(replay.get("hard_risk_violations", 0)) == 0,        "zero_orphan_orders": int(replay.get("orphan_orders", 0)) == 0,    }    return {        "overall_pass": all(gates.values()),        "token_speedup_vs_baseline": round(token_speedup, 2),        "iecu_per_decision": round(iecu_per_dec, 1),        "hold_quality": hold_q,        "noninferiority": ni_res,        "usage_stats": stats,        "gates": gates,    }if __name__ == "__main__":    ap = argparse.ArgumentParser()    ap.add_argument("--ledger", type=Path, default=Path("runtime/usage_ledger.sqlite3"))    ap.add_argument("--replay", type=Path, required=True)    args = ap.parse_args()    report = evaluate_matrix(args.ledger, args.replay)    print(json.dumps(report, indent=2, ensure_ascii=False))    raise SystemExit(0 if report["overall_pass"] else 1)
